local function preciseWait(seconds)
	if not seconds or seconds <= 0 then return end
	local elapsed = 0
	while elapsed < seconds do
		elapsed = elapsed + RunService.Heartbeat:Wait()
	end
end

local function getPlayerSpeed()
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local speedStat = leaderstats:FindFirstChild("Speed") or leaderstats:FindFirstChild("speed")
		if speedStat and speedStat:IsA("ValueBase") then

local cachedGroundObj = nil

local function getGroundObject()
	if cachedGroundObj and cachedGroundObj.Parent then
		return cachedGroundObj
	end
	local objects = workspace:FindFirstChild("__OBJECTS")
	local areas = objects and objects:FindFirstChild("Areas")
	cachedGroundObj = areas and areas:FindFirstChild("Ground")
	return cachedGroundObj
end

local function isPlayerOnGround()
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return false end
	local groundObj = getGroundObject()
	if not groundObj then return false end
	local pos = root.Position
	if groundObj:IsA("Model") then
		local cf, size = groundObj:GetBoundingBox()
		local localPos = cf:PointToObjectSpace(pos)
		return math.abs(localPos.X) <= (size.X * 0.5 + 10)
			and math.abs(localPos.Y) <= (size.Y * 0.5 + 30)
			and math.abs(localPos.Z) <= (size.Z * 0.5 + 10)
	elseif groundObj:IsA("BasePart") then
		local localPos = groundObj.CFrame:PointToObjectSpace(pos)
		local halfSize = groundObj.Size * 0.5
		return math.abs(localPos.X) <= (halfSize.X + 10)
			and math.abs(localPos.Y) <= (halfSize.Y + 30)
			and math.abs(localPos.Z) <= (halfSize.Z + 10)
	end
	return false
end

local function stopCurrentWalking(shouldDropEgg)
	currentWalkToken = currentWalkToken + 1
	isWalkingToTarget = false
	activeTargetSpeed = nil
	setMovementControlsEnabled(true)
	
	task.spawn(function()
		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum = activeHumanoid or (char and char:FindFirstChildOfClass("Humanoid"))
		
		if hum then
			hum:Move(Vector3.zero, false)

		end

local function resolvePosition(targetDestination)
	if typeof(targetDestination) == "Vector3" then
		return targetDestination
	elseif typeof(targetDestination) == "Instance" then
		if targetDestination:IsA("BasePart") then
			return targetDestination.Position
		elseif targetDestination:IsA("Model") then
			return targetDestination:GetPivot().Position
		end
	end
	return nil
end

local function walkDirect(targetPos, speed, threshold, thisToken)
	isWalkingToTarget = true
	local targetSpeed = speed or Settings.WalkSpeed
	if targetSpeed > 500 then targetSpeed = 500 end
	activeTargetSpeed = targetSpeed

	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local hum = activeHumanoid or (char and char:FindFirstChildOfClass("Humanoid"))
	if not root or not hum then 
		isWalkingToTarget = false 
		return false
	end

	hum.WalkSpeed = activeTargetSpeed
	hum:MoveTo(targetPos)

	local targetX, targetZ = targetPos.X, targetPos.Z
	local reqThreshold = threshold or Settings.ArrivalThreshold

	while currentWalkToken == thisToken and isWalkingToTarget and scriptRunning and root and hum do
		hum.WalkSpeed = activeTargetSpeed
		local currentPos = root.Position
		local dx = targetX - currentPos.X
		local dz = targetZ - currentPos.Z
		local distSq = dx * dx + dz * dz
		if distSq <= (reqThreshold * reqThreshold) then
			break
		end
		hum:MoveTo(targetPos)
		RunService.Heartbeat:Wait()
	end

	return currentWalkToken == thisToken and scriptRunning
end

local function walkToTarget(targetDestination, customSpeed, customThreshold)
	stopCurrentWalking(false)
	currentWalkToken = currentWalkToken + 1
	local thisToken = currentWalkToken

	task.spawn(function()
		local finalPos = resolvePosition(targetDestination)
		if not finalPos then return end

		setMovementControlsEnabled(false)

		if isPlayerOnGround() then
			unequipTreadmill()
		end

		local char = player.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		if root then
			local targetZWaypoint = Vector3.new(root.Position.X, root.Position.Y, finalPos.Z)
			if not walkDirect(targetZWaypoint, customSpeed, customThreshold, thisToken) then
				setMovementControlsEnabled(true)
				return
			end
			if not walkDirect(finalPos, customSpeed, customThreshold, thisToken) then
				setMovementControlsEnabled(true)
				return
			end
		end

		if currentWalkToken == thisToken then
			local hum = activeHumanoid or (player.Character and player.Character:FindFirstChildOfClass("Humanoid"))
			local rootPart = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if hum and rootPart then
				hum:Move(Vector3.zero, false)
				hum:MoveTo(rootPart.Position)
			end
			isWalkingToTarget = false
			activeTargetSpeed = nil
			applySpeedRealtime()
			setMovementControlsEnabled(true)
		end
	end)
end

local function applyAntiRagdoll(character, humanoid)
	if not character or not humanoid then return end
	
	humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
	humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
	humanoid:SetStateEnabled(Enum.HumanoidStateType.GettingUp, true)
	humanoid:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
	
	for _, descendant in pairs(character:GetDescendants()) do
		if descendant:IsA("Motor6D") then
			descendant.Enabled = true
		end
	end
end

local function applyZombieDesync(character)
	if not character then return end
	
	local originalHumanoid = character:WaitForChild("Humanoid", 5)
	if not originalHumanoid then return end
	
	local healthScript = character:FindFirstChild("Health")
	if healthScript then healthScript:Destroy() end
	
	originalHumanoid.BreakJointsOnDeath = false
	
	local clonedHumanoid = originalHumanoid:Clone()
	clonedHumanoid.Name = "Humanoid"
	clonedHumanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
	clonedHumanoid.Health = originalHumanoid.MaxHealth
	
	local animator = clonedHumanoid:FindFirstChildOfClass("Animator")
	if not animator then
		animator = Instance.new("Animator")
		animator.Parent = clonedHumanoid
	end
	
	if Settings.JumpAnimationId and Settings.JumpAnimationId ~= "" then
		pcall(function()
			local jumpAnim = Instance.new("Animation")
			jumpAnim.AnimationId = Settings.JumpAnimationId
			local jumpTrack = animator:LoadAnimation(jumpAnim)

			clonedHumanoid.Jumping:Connect(function(isActive)
				if isActive then
					jumpTrack:Play()
				end
			end)
		end)
	end
	
	clonedHumanoid.JumpPower = 50
	clonedHumanoid.JumpHeight = 7.2
	clonedHumanoid.UseJumpPower = true

	originalHumanoid:Destroy()
	clonedHumanoid.Parent = character
	activeHumanoid = clonedHumanoid
	applySpeedRealtime()
	
	applyAntiRagdoll(character, clonedHumanoid)
	
	local camera = workspace.CurrentCamera
	if camera then camera.CameraSubject = clonedHumanoid end
	
	local animateScript = character:FindFirstChild("Animate")
	if animateScript then
		local animateClone = animateScript:Clone()
		animateScript:Destroy()
		animateClone.Parent = character
		animateClone.Disabled = false
	end
end

local TARGET_NAMES = {
	["AssetRandomIdleSound"] = true,
	["AssetWalkSound"] = true,
	["AssetWalkSoundEmitterWeld"] = true
}

local function freezeInstance(instance)
	if instance:IsA("BasePart") then
		instance.Anchored = true
		instance.AssemblyLinearVelocity = Vector3.zero
		instance.AssemblyAngularVelocity = Vector3.zero
	end
	if instance:IsA("AnimationController") or instance:IsA("Humanoid") then
		local animator = instance:FindFirstChildOfClass("Animator")
		if animator then
			for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
				track:Stop()
			end
		end
	end
	if instance:IsA("ParticleEmitter") or instance:IsA("Beam") or instance:IsA("Trail") then
		instance.Enabled = false
	end
end

local function cleanTargetElement(instance)
	local currentAncestor = instance
	while currentAncestor and currentAncestor ~= game do
		if currentAncestor:IsA("Model") and Players:GetPlayerFromCharacter(currentAncestor) then
			return
		end
		currentAncestor = currentAncestor.Parent
	end

	if TARGET_NAMES[instance.Name] then
		pcall(function() instance:Destroy() end)
		return
	end

	if instance:IsA("Animator") or instance:IsA("AnimationController") then
		pcall(function()
			for _, track in ipairs(instance:GetPlayingAnimationTracks()) do
				track:Stop()
			end
			instance:Destroy()
		end)
	elseif instance:IsA("Animation") then
		instance.AnimationId = ""
		pcall(function() instance:Destroy() end)
	elseif instance:IsA("Motor6D") or instance:IsA("Motor") or instance:IsA("Weld") then
		pcall(function() instance.Transform = CFrame.identity end)
	elseif instance:IsA("ParticleEmitter") or instance:IsA("Trail") or instance:IsA("Smoke") or instance:IsA("Fire") or instance:IsA("Sparkles") then
		instance.Enabled = false
		pcall(function() instance:Clear() end)
		pcall(function() instance:Destroy() end)
	elseif instance:IsA("Beam") then
		instance.Enabled = false
		pcall(function() instance:Destroy() end)
	elseif instance:IsA("Atmosphere") or instance:IsA("Clouds") then
		pcall(function() instance:Destroy() end)
	elseif instance:IsA("Texture") or instance:IsA("Decal") then
		local name = instance.Name:lower()
		if name:find("water") or name:find("river") or name:find("flow") or name:find("wave") or name:find("lava") or name:find("fire") then
			instance.Transparency = 1
		end
	elseif instance:IsA("Script") or instance:IsA("LocalScript") then
		local name = instance.Name:lower()
		if name:find("anim") or name:find("pet") or name:find("float") or name:find("bob") or name:find("rotate") or name:find("spin") or name:find("water") or name:find("river") then
			instance.Disabled = true
		end
	end
end

local function toggleFpsBooster(forceState)
	if forceState ~= nil then
		fpsBoosterActive = forceState
	else
		fpsBoosterActive = not fpsBoosterActive
	end

	if fpsBoosterActive then
		pcall(function()
			Lighting.FogStart = 1e10
			Lighting.FogEnd = 1e10
			for _, child in ipairs(Lighting:GetChildren()) do
				if child:IsA("Atmosphere") or child:IsA("Clouds") or child:IsA("PostEffect") then
					child.Enabled = false
				end
			end
		end)

		if Terrain then
			pcall(function()
				Terrain.WaterWaveSpeed = 0
				Terrain.WaterWaveSize = 0
				Terrain.WaterReflectance = 0
				Terrain.WaterTransparency = 1
			end)
		end

		local assetsFolder = workspace:FindFirstChild("ClientRenderedAssets")
		if assetsFolder then
			for _, descendant in ipairs(assetsFolder:GetDescendants()) do
				freezeInstance(descendant)
			end
		end

		for _, instance in ipairs(workspace:GetDescendants()) do
			cleanTargetElement(instance)
		end

		if not _G.FpsBoosterConn then
			_G.FpsBoosterConn = workspace.DescendantAdded:Connect(function(descendant)
				if scriptRunning and fpsBoosterActive then
					if descendant:IsDescendantOf(workspace:FindFirstChild("ClientRenderedAssets")) then
						freezeInstance(descendant)
					end
					cleanTargetElement(descendant)
				end
			end)
			table.insert(scriptConnections, _G.FpsBoosterConn)
		end
	end
end

	end
end)

local fpsCapValue = 0        -- 0 = disabled

local function applyFpsCapLoop()
	task.spawn(function()
		while scriptRunning do
			if fpsCapEnabled and fpsCapValue and fpsCapValue > 0 then
				local frameTime = 1 / fpsCapValue
				local startTime = os.clock()
				RunService.RenderStepped:Wait()
				local elapsed = os.clock() - startTime
				if elapsed < frameTime then
					task.wait(frameTime - elapsed)
				end
			else
				task.wait(0.1)
			end
		end
	end)
end

local function formatNumber(n)
	n = math.floor(n)
	if n >= 1e12 then
		return string.format("%.2fT", n / 1e12)
	elseif n >= 1e9 then
		return string.format("%.2fB", n / 1e9)
	elseif n >= 1e6 then
		return string.format("%.2fM", n / 1e6)
	elseif n >= 1e3 then
		return string.format("%.2fK", n / 1e3)
	end
	return tostring(n)
end

local function getProgressColor(progress)
	progress = math.clamp(progress, 0, 1)
	
	local stops = {
		{pos = 0.00, color = Color3.fromRGB(220, 50, 50)},
		{pos = 0.33, color = Color3.fromRGB(240, 140, 40)},
		{pos = 0.66, color = Color3.fromRGB(240, 220, 40)},
		{pos = 1.00, color = Color3.fromRGB(60, 200, 90)},
	}
	
	for i = 1, #stops - 1 do
		local a, b = stops[i], stops[i + 1]
		if progress >= a.pos and progress <= b.pos then
			local t = (progress - a.pos) / (b.pos - a.pos)
			return a.color:Lerp(b.color, t)
		end
	end
	
	return stops[#stops].color
end

local speedTrackerConn = RunService.Heartbeat:Connect(function()
	if not scriptRunning then return end

	local currentSpeed = getPlayerSpeed()
	currentSpeedLabel.Text = "Speed: " .. formatNumber(currentSpeed)

	local currentMap = nil
	local nextMap = nil
	for _, mapInfo in ipairs(sortedMaps) do
		if currentSpeed >= mapInfo.RequiredSpeed then
			currentMap = mapInfo
		else
			if not nextMap then
				nextMap = mapInfo
			end
		end
	end

	if currentMap then
		currentMapLabel.Text = "At: " .. currentMap.Name
	else
		currentMapLabel.Text = "At: —"
	end

	if nextMap then
		local remaining = nextMap.RequiredSpeed - currentSpeed
		local progress = 0
		if nextMap.RequiredSpeed > 0 then
			progress = math.clamp(currentSpeed / nextMap.RequiredSpeed, 0, 1)
		end

		nextMapLabel.Text = "Next: " .. nextMap.Name
		remainingSpeedLabel.Text = formatNumber(remaining) .. " speed left"

		progressBarFill.Size = UDim2.new(progress, 0, 1, 0)
		progressBarFill.BackgroundColor3 = getProgressColor(progress)
		progressPercentLabel.Text = string.format("%.2f%% → %s", progress * 100, formatNumber(nextMap.RequiredSpeed))
		progressPercentLabel.TextColor3 = getProgressColor(progress)

		currentSpeedLabel.TextColor3 = NINE_BLUE
	else
		nextMapLabel.Text = "Next: —"
		remainingSpeedLabel.Text = "All maps unlocked!"
		progressBarFill.Size = UDim2.new(1, 0, 1, 0)
		progressBarFill.BackgroundColor3 = Color3.fromRGB(60, 200, 90)
		progressPercentLabel.Text = "MAX"
		progressPercentLabel.TextColor3 = Color3.fromRGB(60, 200, 90)
		currentSpeedLabel.TextColor3 = Color3.fromRGB(60, 200, 90)
	end
end)

local function applyFpsCapText(rawText)
	if rawText == "" or rawText == "0" then
		fpsCapEnabled = false
		fpsCapValue = 0
		return 0
	end

	local num = tonumber(rawText)
	if not num then
		fpsCapEnabled = false
		fpsCapValue = 0
		return nil
	end

	num = math.floor(num)
	if num < FPS_MIN then num = FPS_MIN end
	if num > FPS_MAX then num = FPS_MAX end

	fpsCapValue = num
	fpsCapEnabled = true
	return num
end

fpsCapInput:GetPropertyChangedSignal("Text"):Connect(function()
	local result = applyFpsCapText(fpsCapInput.Text)
	if result and result ~= tonumber(fpsCapInput.Text) then
		-- Clamp without spamming (only update text if out of range)
		if tonumber(fpsCapInput.Text) and (tonumber(fpsCapInput.Text) < FPS_MIN or tonumber(fpsCapInput.Text) > FPS_MAX) then
			fpsCapInput.Text = tostring(result)
		end
	end
end)

fpsCapInput.FocusLost:Connect(function()
	local result = applyFpsCapText(fpsCapInput.Text)
	if result == nil or result == 0 then
		fpsCapInput.Text = tostring(FPS_DEFAULT)
		applyFpsCapText(tostring(FPS_DEFAULT))
	else
		fpsCapInput.Text = tostring(result)
	end
end)

local isSearchingServer = false

btnFindLowServer.MouseButton1Click:Connect(function()
	if isSearchingServer then return end
	isSearchingServer = true
	
	btnFindLowServer.Text = "Searching..."
	btnFindLowServer.BackgroundColor3 = Color3.fromRGB(100, 100, 105)

	task.spawn(function()
		local PlaceId = game.PlaceId
		local CurrentJobId = game.JobId
		local URL = string.format("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Asc&limit=100&excludeFullGames=true&_=%d", tostring(PlaceId), os.time())

		local success, response = pcall(function()
			return game:HttpGet(URL)
		end)

		if not success or not response then
			warn("[Server Finder]: Failed to retrieve server list.")
			btnFindLowServer.Text = "Error! Retry"
			btnFindLowServer.BackgroundColor3 = Color3.fromRGB(248, 113, 113)
			preciseWait(2)
			btnFindLowServer.Text = "Find Low Server"
			btnFindLowServer.BackgroundColor3 = NINE_BLUE
			isSearchingServer = false
			return
		end

		local decodeSuccess, decodedData = pcall(function()
			return HttpService:JSONDecode(response)
		end)

		local validServers = {}

		if decodeSuccess and decodedData and decodedData.data then
			for _, server in ipairs(decodedData.data) do
				if typeof(server) == "table" and server.id ~= CurrentJobId and (server.playing or 0) < (server.maxPlayers or 0) then
					table.insert(validServers, server)
				end
			end
		end

		if #validServers > 0 then
			table.sort(validServers, function(a, b)
				if a.playing ~= b.playing then
					return (a.playing or 0) < (b.playing or 0)
				end
				return (a.ping or 999) < (b.ping or 999)
			end)

			local targetServer = validServers[1]
			btnFindLowServer.Text = "Teleporting..."
			
			local tpSuccess, tpError = pcall(function()
				TeleportService:TeleportToPlaceInstance(PlaceId, targetServer.id, Players.LocalPlayer)
			end)

			if not tpSuccess then
				warn("[Server Finder]: Teleport failed - " + tostring(tpError))
				btnFindLowServer.Text = "TP Failed! Retry"
				btnFindLowServer.BackgroundColor3 = Color3.fromRGB(248, 113, 113)
				preciseWait(2)
			end
		else
			warn("[Server Finder]: No alternative low-player servers found.")
			btnFindLowServer.Text = "No Server Found"
			btnFindLowServer.BackgroundColor3 = Color3.fromRGB(251, 191, 36)
			preciseWait(2)
		end

		btnFindLowServer.Text = "Find Low Server"
		btnFindLowServer.BackgroundColor3 = NINE_BLUE
		isSearchingServer = false
	end)
end)

local oldNamecall; oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
	local method = getnamecallmethod()
	local args = {...}

	if scriptRunning and method == "InvokeServer" and self.Name == "RF/EggWorld/AskFieldEggCarry" then
		local result = oldNamecall(self, unpack(args))
		
		if result == true then
			if toggleStateV1 then
				task.spawn(function()
					preciseWait(Settings.HookDelay)
					if toggleStateV1 and scriptRunning then
						local targetPos = Settings.BaseDropPosition
						if targetPos then
							walkToTarget(targetPos, Settings.MovementSpeed, 3)
							repeat RunService.Heartbeat:Wait() until not isWalkingToTarget or not scriptRunning
							
							if scriptRunning and toggleStateV1 then
								preciseWait(Settings.DropEggDelay)
								dropEggServer()
							end
						end
					end
				end)
			end
		else
			stopCurrentWalking(true)
		end

		return result
	end

	return oldNamecall(self, ...)
end))

local function makePromptInstant(prompt)
	if prompt:IsA("ProximityPrompt") then
		prompt.HoldDuration = 0
		prompt.MaxActivationDistance = math.max(prompt.MaxActivationDistance, 10)
		prompt.RequiresLineOfSight = false
	end
end

end)

local debris = workspace:WaitForChild("__DEBRIS")
local function processChild(child)
	local hitbox = child:FindFirstChild("Hitbox")
	if hitbox then
		local ti = hitbox:FindFirstChild("TouchInterest") or hitbox:FindFirstChildOfClass("TouchInterest")
		if ti then ti:Destroy() end
		hitbox.ChildAdded:Connect(function(newChild)
			if newChild:IsA("TouchInterest") or newChild.Name == "TouchInterest" then
				RunService.Heartbeat:Wait()
				newChild:Destroy()
			end
		end)
	end
end

for _, child in ipairs(debris:GetChildren()) do processChild(child) end
debris.ChildAdded:Connect(processChild)
